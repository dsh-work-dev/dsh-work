cmake_minimum_required(VERSION 3.21)
project(dsh_cef_broker)
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Formal native transport is currently Windows/Linux only")
endif()
set(USE_SANDBOX ON CACHE BOOL "CEF sandbox" FORCE)
set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
file(READ "${CEF_ROOT}/include/cef_version.h" CEF_VERSION_HEADER)
if(NOT CEF_VERSION_HEADER MATCHES "156.0.15\\+gcf671c0\\+chromium-156.0.8078.80")
  message(FATAL_ERROR "Private transport candidate requires the validated CEF156 SDK")
endif()
add_subdirectory(${CEF_LIBCEF_DLL_WRAPPER_PATH} libcef_dll_wrapper)
ADD_LOGICAL_TARGET(libcef_lib "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
set(CEF_TARGET dsh_cef_broker)
SET_CEF_TARGET_OUT_DIR()
if(WIN32)
add_library(${CEF_TARGET} SHARED broker.cc)
SET_LIBRARY_TARGET_PROPERTIES(${CEF_TARGET})
target_link_libraries(${CEF_TARGET} libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS} bcrypt ole32 shell32 uuid)
COPY_SINGLE_FILE(${CEF_TARGET} ${CEF_BINARY_DIR}/bootstrap.exe ${CEF_TARGET_OUT_DIR}/${CEF_TARGET}.exe)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(XCB REQUIRED IMPORTED_TARGET xcb xcb-res)
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  set_target_properties(libcef_dll_wrapper PROPERTIES POSITION_INDEPENDENT_CODE ON)
  add_executable(${CEF_TARGET} broker.cc)
  SET_EXECUTABLE_TARGET_PROPERTIES(${CEF_TARGET})
  target_link_libraries(${CEF_TARGET} libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS} PkgConfig::XCB OpenSSL::Crypto)
  set_target_properties(${CEF_TARGET} PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()
target_compile_features(${CEF_TARGET} PRIVATE cxx_std_20)
COPY_FILES(${CEF_TARGET} "${CEF_BINARY_FILES}" "${CEF_BINARY_DIR}" "${CEF_TARGET_OUT_DIR}")
COPY_FILES(${CEF_TARGET} "${CEF_RESOURCE_FILES}" "${CEF_RESOURCE_DIR}" "${CEF_TARGET_OUT_DIR}")
if(WIN32)
SET_LPAC_ACLS(${CEF_TARGET})

add_executable(dsh_presenter_window_test presentation_window_test.cc)
target_compile_features(dsh_presenter_window_test PRIVATE cxx_std_20)
target_link_libraries(dsh_presenter_window_test user32)
endif()
