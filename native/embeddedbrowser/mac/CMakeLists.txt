cmake_minimum_required(VERSION 3.21)
project(dsh_owned_cef_mac LANGUAGES C CXX OBJCXX)
if(NOT APPLE OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
  message(FATAL_ERROR "The formal CEF component requires native macOS arm64")
endif()
set(USE_SANDBOX ON CACHE BOOL "CEF sandbox" FORCE)
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
file(READ "${CEF_ROOT}/include/cef_version.h" CEF_VERSION_HEADER)
if(NOT CEF_VERSION_HEADER MATCHES "156.0.15\\+gcf671c0\\+chromium-156.0.8078.80")
  message(FATAL_ERROR "Approved CEF156 SDK required")
endif()
add_library(owned_official_wrapper STATIC IMPORTED)
set_target_properties(owned_official_wrapper PROPERTIES IMPORTED_LOCATION "${CEF_WRAPPER_LIBRARY}")
set(BROKER "${CMAKE_CURRENT_SOURCE_DIR}/../broker.cc")
set_source_files_properties("${BROKER}" PROPERTIES LANGUAGE OBJCXX)
set(MACH "${CMAKE_CURRENT_SOURCE_DIR}/../../../internal/localipc/mach_peer_darwin.c")
add_executable(dsh_cef_broker "${BROKER}" main.mm platform_mac.mm "${MACH}")
add_executable(dsh_cef_helper helper_main.cc platform_mac.mm "${MACH}")
foreach(target dsh_cef_broker dsh_cef_helper)
  SET_EXECUTABLE_TARGET_PROPERTIES(${target})
  target_include_directories(${target} PRIVATE "${CEF_ROOT}")
  target_link_libraries(${target} owned_official_wrapper ${CEF_STANDARD_LIBS} "-framework Foundation" "-framework Cocoa")
  target_compile_features(${target} PRIVATE cxx_std_20)
  set_target_properties(${target} PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES OBJCXX_EXTENSIONS NO RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
endforeach()
