# Included from the official CEF standard SDK build; use its maintained wrapper.
if(OS_MAC)
  set(BRIDGE_SOURCE bridge.mm)
else()
  set(BRIDGE_SOURCE bridge_linux.cc)
endif()
add_library(browser_lab_bridge SHARED ${BRIDGE_SOURCE})
set_target_properties(libcef_dll_wrapper PROPERTIES POSITION_INDEPENDENT_CODE ON)
SET_LIBRARY_TARGET_PROPERTIES(browser_lab_bridge)
target_include_directories(browser_lab_bridge PRIVATE "${CEF_ROOT}")
set_target_properties(browser_lab_bridge PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lab")
if(OS_MAC)
  target_link_libraries(browser_lab_bridge libcef_dll_wrapper ${CEF_STANDARD_LIBS} "-framework Cocoa")
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4)
  # SDK sample logical imports are directory-scoped; create our own import.
  ADD_LOGICAL_TARGET(browser_lab_cef "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
  target_link_libraries(browser_lab_bridge PkgConfig::GTK4 X11)
  add_executable(browser_lab_broker broker_linux.cc)
  SET_EXECUTABLE_TARGET_PROPERTIES(browser_lab_broker)
  target_link_libraries(browser_lab_broker browser_lab_cef libcef_dll_wrapper ${CEF_STANDARD_LIBS})
  set_target_properties(browser_lab_broker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lab" INSTALL_RPATH "$ORIGIN" BUILD_WITH_INSTALL_RPATH TRUE)
  # Load the staged runtime beside the bridge, not SDK/Release (which lacks
  # the staged ICU/resources). Match the official cefsimple runtime layout.
  set_target_properties(browser_lab_bridge PROPERTIES INSTALL_RPATH "$ORIGIN" BUILD_WITH_INSTALL_RPATH TRUE)
endif()
